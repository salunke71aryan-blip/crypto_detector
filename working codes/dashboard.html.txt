    <script>
        function switchView(targetTab) {
            document.getElementById("btnTracker").classList.remove("active");
            document.getElementById("btnMetrics").classList.remove("active");
            document.getElementById("btnReports").classList.remove("active");
            
            document.getElementById("trackerOverviewView").classList.add("hidden-view");
            document.getElementById("metricsAnalyticsView").classList.add("hidden-view");
            document.getElementById("caseEvidenceReportsView").classList.add("hidden-view");
            
            if (targetTab === 'tracker') {
                document.getElementById("btnTracker").classList.add("active");
                document.getElementById("trackerOverviewView").classList.remove("hidden-view");
            } else if (targetTab === 'metrics') {
                document.getElementById("btnMetrics").classList.add("active");
                document.getElementById("metricsAnalyticsView").classList.remove("hidden-view");
            } else if (targetTab === 'reports') {
                document.getElementById("btnReports").classList.add("active");
                document.getElementById("caseEvidenceReportsView").classList.remove("hidden-view");
                fetchForensicReportFile(); 
            }
        }

        function fetchForensicReportFile() {
            fetch('/get_report')
            .then(res => res.json())
            .then(data => {
                if (data.success) {
                    document.getElementById("reportDisplayContainer").innerText = data.content;
                } else {
                    document.getElementById("reportDisplayContainer").innerText = "Error: " + data.error;
                }
            })
            .catch(err => {
                document.getElementById("reportDisplayContainer").innerText = "Failed to load report from server backend.";
            });
        }

        function triggerLiveTrace() {
            const walletValue = document.getElementById("walletInput").value.trim();
            
            fetch('/trace', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ wallet: walletValue })
            })
            .then(response => response.json())
            .then(data => {
                if(data.success) {
                    document.getElementById("statVasp").innerText = data.exchange;
                    document.getElementById("statRisk").innerText = data.risk;
                    document.getElementById("statHops").innerText = data.hops + " Hops";
                    document.getElementById("statFreeze").innerText = data.freeze;
                    
                    document.getElementById("mapFrame").src = "/map_frame?t=" + new Date().getTime();
                } else {
                    alert("Verification Error: " + data.error);
                }
            })
            .catch(err => console.error("Communication broke:", err));
        }
    </script>
