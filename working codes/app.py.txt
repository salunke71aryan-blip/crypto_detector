def calculate_wallet_risk(wallet_id):
    profile = wallet_profiles.get(wallet_id, {"tx_count": random.randint(35, 150), "hours_active": random.randint(1, 5), "mixers_touched": random.choice([True, False]), "peeling_detected": True})
    score = 10
    if profile["tx_count"] > 30 and profile["hours_active"] < 12: score += 30
    if profile["mixers_touched"]: score += 40
    if profile["peeling_detected"]: score += 20
    score = min(score, 100)
    return "CRITICAL" if score >= 80 else "HIGH" if score >= 50 else "MEDIUM" if score >= 25 else "LOW"

def generate_graph_html(trace_results):
    net = Network(height="450px", width="100%", bgcolor="#ffffff", font_color="#000000", directed=True, cdn_resources='in_line')
    all_nodes = set([e["from"] for e in trace_results["edges"]] + [e["to"] for e in trace_results["edges"]])

    for node in all_nodes:
        display_label = node[:6] + "..." + node[-4:] if len(node) > 15 else node
        if node == trace_results['path'][0]:
            net.add_node(node, label=f"VICTIM\n({display_label})", color="#22c55e", size=25)
        elif node == trace_results.get("target_wallet"):
            net.add_node(node, label=f"EXCHANGE\n({trace_results['exchange']})", color="#a855f7", size=32, shape="diamond")
        else:
            net.add_node(node, label=f"SUSPECT\n({display_label})", color="#ef4444" if calculate_wallet_risk(node) == "CRITICAL" else "#38bdf8", size=20)

    for edge in trace_results["edges"]:
        net.add_edge(edge["from"], edge["to"], label=f"{edge['amount']} Crypto")

    net.write_html(os.path.join("templates", "dynamic_crypto_map.html"))

def generate_investigation_report(trace_results):
    filename = "investigation_report.md"
    report_content = f"""# ⚖️ CRYPTO FRAUD INTELLIGENCE REPORT
**Generated By:** Tech Titans Forensic Core
**Status:** EVIDENCE COMPILED (READY FOR LEA ACTION)  

* **Suspect Input Source:** `{trace_results['path'][0]}`
* **Attributed Endpoint VASP:** **{trace_results['exchange']}**
* **Target Deposit Wallet:** `{trace_results['target_wallet']}`
* **Total Network Hops Traversed:** {len(trace_results['edges'])} Hops

---
### Definitive Audit Trail Ledger:
"""
    for i, edge in enumerate(trace_results["edges"], 1):
        report_content += f"- Hop #{i}: From `{edge['from']}` -> To `{edge['to']}` | Amount: **{edge['amount']} Crypto** | TxID: `{edge['txid']}`\n"
        
    with open(filename, "w", encoding="utf-8") as f:
        f.write(report_content)

    generate_graph_html(initial_trace)

@app.route('/trace', methods=['POST'])
def run_trace():
    target_wallet = request.json.get("wallet", "").strip()
    if not target_wallet:
        return jsonify({"success": False, "error": "Empty wallet address field input string."})
        
    trace_output = trace_stolen_funds(target_wallet)
    generate_graph_html(trace_output)
    generate_investigation_report(trace_output)
    
    return jsonify({
        "success": True,
        "exchange": trace_output['exchange'],
        "risk": "CRITICAL (95%)" if target_wallet.startswith("bc1q") else "CRITICAL (90%)",
        "hops": len(trace_output['edges']),
        "freeze": trace_output['freeze']
    })

@app.route('/get_report')
def get_report():
    if os.path.exists("investigation_report.md"):
        with open("investigation_report.md", "r", encoding="utf-8") as f:
            return jsonify({"success": True, "content": f.read()})
    return jsonify({"success": False, "error": "Report file not compiled yet."})
